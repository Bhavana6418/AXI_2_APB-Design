`timescale 1ns/1ps 
module pulse_sync(input clk,rstn,d,output p);
	reg meta1,meta2,meta3;

	always@(posedge clk or negedge rstn)
	begin
		if(!rstn)
		begin
			meta1 <= 1'b0;
			meta2 <= 1'b0;
			meta3 <= 1'b0;
		end
		else
		begin
			meta1 <= d;
			meta2 <= meta1;
			meta3 <= ~meta2;
		end
	end

	assign p = meta2 & meta3;

endmodule
