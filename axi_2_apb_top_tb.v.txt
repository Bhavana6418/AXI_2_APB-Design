always #5 Aclk_i = ~Aclk_i;
always #10 Pclk_i = ~Pclk_i;

task reset;
	begin
		Aresetn_i = 1'b0;
		Presetn_i = 1'b0;
		#40;
		Aresetn_i = 1'b1;
		Presetn_i = 1'b1;
	end
endtask

task initialize;
	begin
		// initialize clk
		Pclk_i = 0;
		Aclk_i = 0;

		// reset signals
		Presetn_i = 1;
		Aresetn_i = 1;

		// initialize all inputs to default
		pready_i = 0;
		pslverr_i = 0;
		{prdata_i_0,prdata_i_1,prdata_i_2,prdata_i_3,prdata_i_4,prdata_i_5,prdata_i_6,prdata_i_7,prdata_i_8,prdata_i_9} = 0;
		{Awid_i,Awaddr_i,Awlen_i,Awsize_i,Awburst_i,Awprot_i,Awvalid_i} = 0;
		{Wdata_i,Wstrb_i,Wlast_i,Wvalid_i} = 0;
		Bready_i = 0;
		{Arid_i,Araddr_i,Arlen_i,Arsize_i,Arburst_i,Arprot_i,Arvalid_i} = 0;
		Rready_i = 0;
		#10;
	end
endtask

//------------------- task to send axi_write_addr and cntrl info ---------------------//

task axi_write_address(input [IW-1:0] aw_id, input [AW-1:0] aw_addr, input [LW-1:0] aw_len, input [2:0] aw_size,input [1:0] aw_burst, input [2:0] aw_prot);
	begin
		@(negedge Aclk_i)
		Awid_i = aw_id;
		Awaddr_i = aw_addr;
		Awlen_i = aw_len;
		Awsize_i = aw_size;
		Awburst_i = aw_burst;
		Awprot_i = aw_prot;
		Awvalid_i = 1'b1;
		wait(Awready_o)
		@(negedge Aclk_i)
		Awvalid_i = 1'b0;
	end
endtask

//------------------- task to send axi_write_data strobe ---------------------//

task axi_write_data(input [LW-1:0] len);
	begin
		for(i=0; i<=len; i=i+1)
		begin
			@(negedge Aclk_i)
			Wvalid_i = 1'b1;
			Wdata_i = $random;
			Wstrb_i = 4'b1111;
			if(i == len)
				Wlast_i = 1'b1;
			else
				Wlast_i = 1'b0;
			wait(Wready_o)
			@(negedge Aclk_i)
			Wvalid_i = 1'b0;
			Wlast_i = 1'b0;
		end
	end
endtask

//------------------- task to receive axi_write response ---------------------//

task axi_write_response;
	begin
		@(negedge Aclk_i)
		Bready_i = 1'b1;
		wait(Bvalid_o)
		@(posedge Aclk_i)
		Bready_i = 1'b0;
	end
endtask

//------------------- task to send axi_read_addr and cntrl info ---------------------//

task axi_read_address(input [IW-1:0] ar_id, input [AW-1:0] ar_addr, input [LW-1:0] ar_len, input [2:0] ar_size, input [1:0] ar_burst, input [2:0] ar_prot);
	begin
		@(negedge Aclk_i)
		Arid_i = ar_id;
		Araddr_i = ar_addr;
		Arlen_i = ar_len;
		Arsize_i = ar_size;
		Arburst_i = ar_burst;
		Arprot_i = ar_prot;
		Arvalid_i = 1'b1;
		wait(Arready_o)
		@(negedge Aclk_i)
		Arvalid_i = 1'b0;
	end
endtask

//------------------- task to receive axi_read_data ---------------------//

task axi_read_data(input [LW-1:0] r_len);
	begin
		for(j=0; j<=r_len; j=j+1)
		begin
			@(negedge Aclk_i)
		//	@(negedge Aclk_i)
			Rready_i = 1'b1;
			wait(Rvalid_o)
			@(posedge Aclk_i)
		//	@(negedge Aclk_i)
			Rready_i = 1'b0;
		end
	end
endtask

//------------------- task to send or receive the apb_data based on pwrite and psel  ---------------------//

task apb_operation(input [LW-1:0] len, input resp);
	begin
		for(k=0; k<=len; k=k+1)
		begin
			@(negedge Pclk_i)
			wait(Pselx_o)
				begin
					if(pwrite_o)
					begin
						case(Pselx_o)
							    10'b00_0000_0001 :begin
								    	      wait(penable_o)
									      @(negedge Pclk_i)
								              begin
								              pready_i = 10'b00_0000_0001;
									      pslverr_i = {9'b0,resp};
								              end
								              end

							    10'b00_0000_0010 :begin
								              wait(penable_o)
									      @(negedge Pclk_i)
								              begin
								              pready_i = 10'b00_0000_0010;
									      pslverr_i = {8'b0,resp,1'b0};
								              end
								              end

							    10'b00_0000_0100 :begin
								    	      wait(penable_o)
									      @(negedge Pclk_i)
								              begin

								              pready_i = 10'b00_0000_0100;
									      pslverr_i = {7'b0,resp,2'b0};
								              end
								              end

							    10'b00_0000_1000 :begin
								    	      wait(penable_o)
									      @(negedge Pclk_i)
								              begin

								              pready_i = 10'b00_0000_1000;
									      pslverr_i = {6'b0,resp,3'b0};
								              end
								      	      end

							   10'b00_0001_0000 : begin
								              wait(penable_o)
									      @(negedge Pclk_i)
								              begin

								              pready_i = 10'b00_0001_0000;
									      pslverr_i = {5'b0,resp,4'b0};
								              end
								              end

							   10'b00_0010_0000 : begin
								   	      wait(penable_o)
									      @(negedge Pclk_i)
								              begin

								              pready_i = 10'b00_0010_0000;
									      pslverr_i = {4'b0,resp,5'b0};
								              end
								              end

							  10'b00_0100_0000 : begin
								  	      wait(penable_o)
									      @(negedge Pclk_i)
								              begin

								              pready_i = 10'b00_0100_0000;
									      pslverr_i = {3'b0,resp,6'b0};
								              end
								              end

						          10'b00_1000_0000 : begin
								  	      wait(penable_o)
									      @(negedge Pclk_i)
								              begin

								              pready_i = 10'b00_1000_0000;
									      pslverr_i = {2'b0,resp,7'b0};
								              end
								              end

						          10'b01_0000_0000 : begin
								  	      wait(penable_o)
									      @(negedge Pclk_i)
								              begin

								              pready_i = 10'b01_0000_0000;
									      pslverr_i = {1'b0,resp,8'b0};
								              end
								      	      end

							  10'b10_0000_0000 : begin
								  	      wait(penable_o)
									      @(negedge Pclk_i)
								              begin

								              pready_i = 10'b1000000000;
									      pslverr_i = {resp,9'b0};
								              end
								              end

							  default: 	      begin
								              pready_i = 10'b00_0000_0000;
									      pslverr_i = 10'b0;
								              end
						endcase
						wait(pwrite_o && penable_o)
						@(negedge Pclk_i)
						pready_i = 10'b0;
					end
					else
					begin
						case(Pselx_o)
							    10'b00_0000_0001 : begin
								    	      wait(penable_o)
									      @(negedge Pclk_i)
								              begin
								              pready_i = 10'b00_0000_0001;
									      pslverr_i = {9'b0,resp};
									      prdata_i_0 = $random;
								              end
								      		end

							    10'b00_0000_0010 : begin
								    		wait(penable_o)
									      @(negedge Pclk_i)
								              begin
								              pready_i = 10'b00_0000_0010;
									      pslverr_i = {8'b0,resp,1'b0};
									      prdata_i_1 = $random;
								              end
								      		end

							    10'b00_0000_0100 : begin
								    		wait(penable_o)
									      @(negedge Pclk_i)
								              begin
								              pready_i = 10'b00_0000_0100;
									      pslverr_i = {7'b0,resp,2'b0};
									      prdata_i_2 = $random;
								              end
								      		end

							    10'b0000001000 : begin
								    		wait(penable_o)
									      @(negedge Pclk_i)
								              begin
								              pready_i = 10'b00_00001000;
									      pslverr_i = {6'b0,resp,3'b0};
									      prdata_i_3 = $random;
								              end
									       end

							   10'b00_0001_0000 : begin
								   		wait(penable_o)
									      @(negedge Pclk_i)
								              begin
								              pready_i = 10'b00_0001_0000;
									      pslverr_i = {5'b0,resp,4'b0};
									      prdata_i_4 = $random;
								              end
								      		end

							   10'b00_0010_0000 : begin
								   		wait(penable_o)
									      @(negedge Pclk_i)
								              begin
								              pready_i = 10'b00_0010_0000;
									      pslverr_i = {4'b0,resp,5'b0};
									      prdata_i_5 = $random;
								              end
								      		end

							   10'b00_0100_0000 : begin
								   		wait(penable_o)
									      @(negedge Pclk_i)
								              begin
								              pready_i = 10'b00_0100_0000;
									      pslverr_i = {3'b0,resp,6'b0};
									      prdata_i_6 = $random;
								              end
									      	end

						          10'b00_1000_0000 : begin
								  		wait(penable_o)
									      @(negedge Pclk_i)
								              begin
								              pready_i = 10'b00_1000_0000;
									      pslverr_i = {2'b0,resp,7'b0};
									      prdata_i_7 = $random;
								              end
								      		end

						         10'b01_0000_0000 : begin
								 		wait(penable_o)
									      @(negedge Pclk_i)
								              begin
								              pready_i = 10'b01_0000_0000;
									      pslverr_i = {1'b0,resp,8'b0};
									      prdata_i_8 = $random;
								              end
								      		end

							  10'b10_0000_0000 : begin
								  		wait(penable_o)
									      @(negedge Pclk_i)
								              begin
								              pready_i = 10'b10_0000_0000;
									      pslverr_i = {resp,9'b0};
									      prdata_i_9 = $random;
								              end
									      	end

							  default: 	      begin
								              pready_i = 10'b00_0000_0001;
									      pslverr_i = 10'b0;
								              end
						endcase
						wait(!pwrite_o && penable_o)
						@(negedge Pclk_i)
						pready_i = 10'b0;
				end
			end
		end
	end
endtask

//-------------------------task to send cntrl info --------------------//

task axi_write_addr_csr(input [IW-1:0] aw_id, input [2:0] aw_prot);
	begin
		@(negedge Aclk_i)
		Awid_i = aw_id;
		Awaddr_i = 32'h4000_A000;
		Awlen_i = 4'h0;
		Awsize_i = 3'h0;
		Awburst_i = 2'h0;
		Awprot_i = aw_prot;
		Awvalid_i = 1'b1;
		wait(Awready_o)
		@(negedge Aclk_i)
		Awvalid_i = 1'b0;
	end
endtask

//-------------------------task to send the axi write data to csr --------------------//

task axi_write_data_csr(input [DW-1:0] data);
	begin
			@(negedge Aclk_i)
			Wvalid_i = 1'b1;
			Wdata_i = data;
			Wstrb_i = 4'b1111;
			Wlast_i = 1'b1;
			wait(Wready_o)
			@(negedge Aclk_i)
			Wvalid_i = 1'b0;
			Wlast_i = 1'b0;
	end
endtask
