	begin
		 {PSELx,PWRITE,PENABLE,PPROT,PADDR,PWDATA,PSTRB}=0;

task write (input [31:0] addr);
begin
	PSELx = 1'b1;
	PWRITE = 1'b1;
	PPROT = 3'b1;
	PADDR = addr;
	PWDATA = $random;
	PSTRB = 4'b1111;
	pslverr = 0;
	PENABLE = 1'b1;

	pready = 1;
end
endtask

begin	
	PSELx = 1'b1;
	PWRITE = 1'b0;
	PPROT = 3'b1;
	PADDR = addr;
	PWDATA = $random;
	PSTRB = 4'b1111;
	pslverr = 0;
	PENABLE = 1'b1;
	pready = 1;

	end
endtask
