		          reg raddr_fe;

		          wire rdata_wen;
		          reg rdata_ff;

		          wire [1:0] wr_resp_2_axi;

		          wire PSEL,PENABLE;
		          wire [AW-1:0] PADDR;

		          wire PWRITE;

		          wire [3:0] PSTRB;

	PCLK = 1'b0;
	forever 
	#5 PCLK = ~PCLK;

task initialize;
	begin
		awaddr_ctrl=0;
		waddr_fe=1;
		wdata_strb=0;
		data_fe=1;
		araddr_ctrl=0;
		raddr_fe=1;
		rdata_ff=1;
		access_ratio=0;
		PREADY=0;
		PRDATA=0;
		PSLVERR=0;
		PSLVERR=0;
		decode_err=0;

		decode_err=0;
	end
endtask

task reset;
	begin
		@(negedge PCLK)
		PRESETn = 1'b0;
		@(negedge PCLK)
		PRESETn = 1'b1;
	end
endtask

task write(input last);
	begin
		@(negedge PCLK)
		awaddr_ctrl = {3'd1,4'd3,32'd586,4'd2};
		waddr_fe = 1'b0;
		// data = last,wstrb,wdata
		wdata_strb={last,4'd1,32'd50};
		data_fe = 1'b0;
		
		@(negedge PCLK)
		@(negedge PCLK)
		PREADY = 1;

		@(negedge PCLK)
		waddr_fe = 1'b1;
		data_fe = 1'b1;
		PREADY = 0;
	end
endtask
