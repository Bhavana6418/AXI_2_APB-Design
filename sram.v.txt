`timescale 1ns/1ps 
module sram #(parameter AW=32,DW=32)
	     (input wclk,rclk,wen,ren,
	      input [AW-1:0] waddr,raddr,
	      input [DW-1:0] wdata,
      	      output reg [DW-1:0] rdata);

 reg [DW-1:0] mem[(2**AW)-1:0];

 always@(posedge wclk)
 begin
	 if(wen)
		 mem[waddr]<=wdata;
	 else
		 mem[waddr]<=mem[waddr];
 end

 always@(posedge rclk)
 begin
	 if(ren)
		 rdata<=mem[raddr];
	 else
		 rdata<=rdata;
 end

endmodule
