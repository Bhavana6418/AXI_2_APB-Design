`timescale 1ns/1ps 
module sync_sram #(parameter AW=3,DW=8,DEPTH=8)
	     	  (input clk,wen,ren,
	    	   input [AW-1:0] waddr,raddr,
	       	   input [DW-1:0] wdata,
      	      	   output reg [DW-1:0] rdata);

 reg [DW-1:0] mem[DEPTH-1:0];

 always@(posedge clk)
 begin
	 if(wen)
		 mem[waddr]<=wdata;
	 else
		 mem[waddr]<=mem[waddr];
 end

 always@(posedge clk)
 begin
	 if(ren)
		 rdata<=mem[raddr];
	 else
		 rdata<=rdata;
 end

endmodule
